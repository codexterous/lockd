cmake_minimum_required(VERSION 3.24)
project(source_paths LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LOCKD_TEST_PREFIX_MAP "Map fixture source paths relative to its root" OFF)
if (MSVC)
    add_compile_options(/EHsc /utf-8 /W4 /WX)
else ()
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif ()
if (CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
    # A package build can remap the enclosing checkout through inherited flags. Scope
    # __FILE__ to this fixture's source tree so the ordinary and mapped cases stay distinct.
    set(mapped_root "${PROJECT_SOURCE_DIR}")
    if (LOCKD_TEST_PREFIX_MAP)
        set(mapped_root .)
    endif ()
    add_compile_options("-ffile-prefix-map=${PROJECT_SOURCE_DIR}=${mapped_root}")
endif ()

add_subdirectory(nested "${CMAKE_BINARY_DIR}/separate objects")
add_executable(check main.cpp)
target_include_directories(check PRIVATE "${LOCKD_INCLUDE_DIR}" "${PROJECT_SOURCE_DIR}/include")
target_compile_definitions(check PRIVATE LOCKD_SOURCE_ROOT="${PROJECT_SOURCE_DIR}/")
target_link_libraries(check PRIVATE paths)

add_executable(sdk_check sdk_main.cpp)
target_include_directories(sdk_check PRIVATE "${LOCKD_SDK_INCLUDE_DIR}")
add_executable(constant_check constant_check.cpp)
target_include_directories(constant_check PRIVATE "${LOCKD_INCLUDE_DIR}")
